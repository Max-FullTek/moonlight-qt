get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(DEPS "${REPO_ROOT}/libs/windows")
if(NOT EXISTS "${CURL_SOURCE_DIRECTORY}/CMakeLists.txt")
    message(FATAL_ERROR "Provide verified curl 8.23.0 source with CURL_SOURCE_DIRECTORY")
endif()
set(OPENSSL_INCLUDE_DIR "${DEPS}/include/x64")
set(OPENSSL_SSL_LIBRARY "${DEPS}/lib/x64/libssl.lib")
set(OPENSSL_CRYPTO_LIBRARY "${DEPS}/lib/x64/libcrypto.lib")
set(SSL_EAY_DEBUG "${DEPS}/lib/x64/libssl.lib")
set(SSL_EAY_RELEASE "${DEPS}/lib/x64/libssl.lib")
set(LIB_EAY_DEBUG "${DEPS}/lib/x64/libcrypto.lib")
set(LIB_EAY_RELEASE "${DEPS}/lib/x64/libcrypto.lib")
find_package(OpenSSL REQUIRED)
set(BUILD_SHARED_LIBS OFF)
set(BUILD_STATIC_LIBS ON)
set(BUILD_CURL_EXE OFF)
set(BUILD_TESTING OFF)
set(BUILD_LIBCURL_DOCS OFF)
set(BUILD_MISC_DOCS OFF)
set(CURL_USE_OPENSSL ON)
set(CURL_USE_SCHANNEL OFF)
set(CURL_USE_LIBPSL OFF)
set(CURL_USE_LIBSSH2 OFF)
set(CURL_USE_PKGCONFIG OFF)
set(USE_NGHTTP2 OFF)
set(CURL_ZLIB OFF)
set(CURL_BROTLI OFF)
set(CURL_ZSTD OFF)
add_subdirectory("${CURL_SOURCE_DIRECTORY}" curl)
add_library(lite-native-host-http STATIC "${REPO_ROOT}/app/backend/nativehttptransport.cpp")
target_compile_features(lite-native-host-http PUBLIC cxx_std_17)
target_compile_options(lite-native-host-http PRIVATE /utf-8 /W3 /WX)
target_include_directories(lite-native-host-http PUBLIC "${REPO_ROOT}/app")
target_link_libraries(lite-native-host-http PUBLIC CURL::libcurl OpenSSL::SSL OpenSSL::Crypto)
