cmake_minimum_required(VERSION 3.24)
project(native_parser_capability LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This capability probe requires Windows x64 MSVC")
endif()
foreach(required IN ITEMS EXPAT_SOURCE_DIRECTORY JSON_SOURCE_DIRECTORY)
    if(NOT IS_DIRECTORY "${${required}}")
        message(FATAL_ERROR "Prepare the verified parser releases before configuring")
    endif()
endforeach()
set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_compile_options(/utf-8 /Zi /guard:cf /guard:ehcont)
add_link_options(/DEBUG /guard:cf /guard:ehcont /CETCOMPAT)
add_subdirectory("${EXPAT_SOURCE_DIRECTORY}" expat)
add_executable(native_parser_probe native.cpp)
target_compile_features(native_parser_probe PRIVATE cxx_std_17)
target_compile_options(native_parser_probe PRIVATE /W3 /WX)
target_include_directories(native_parser_probe SYSTEM PRIVATE "${JSON_SOURCE_DIRECTORY}/single_include")
target_link_libraries(native_parser_probe PRIVATE expat::expat)
option(VERIFY_QT_PARSER "Build a Qt reference for compatibility comparison" OFF)
if(VERIFY_QT_PARSER)
    find_package(Qt6 REQUIRED COMPONENTS Core)
    add_executable(qt_parser_reference qt.cpp)
    target_compile_features(qt_parser_reference PRIVATE cxx_std_17)
    target_compile_options(qt_parser_reference PRIVATE /W3 /WX)
    target_link_libraries(qt_parser_reference PRIVATE Qt6::Core)
endif()
