cmake_minimum_required(VERSION 3.24)
project(native_host_http_capability LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This probe requires Windows x64 MSVC")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(DEPS "${REPO_ROOT}/libs/windows")
if(NOT EXISTS "${CURL_SOURCE_DIRECTORY}/CMakeLists.txt")
    message(FATAL_ERROR "Provide verified curl 8.23.0 source with CURL_SOURCE_DIRECTORY")
endif()
set(OPENSSL_INCLUDE_DIR "${DEPS}/include/x64" CACHE PATH "" FORCE)
set(OPENSSL_SSL_LIBRARY "${DEPS}/lib/x64/libssl.lib" CACHE FILEPATH "" FORCE)
set(OPENSSL_CRYPTO_LIBRARY "${DEPS}/lib/x64/libcrypto.lib" CACHE FILEPATH "" FORCE)
set(SSL_EAY_DEBUG "${DEPS}/lib/x64/libssl.lib" CACHE FILEPATH "" FORCE)
set(SSL_EAY_RELEASE "${DEPS}/lib/x64/libssl.lib" CACHE FILEPATH "" FORCE)
set(LIB_EAY_DEBUG "${DEPS}/lib/x64/libcrypto.lib" CACHE FILEPATH "" FORCE)
set(LIB_EAY_RELEASE "${DEPS}/lib/x64/libcrypto.lib" CACHE FILEPATH "" FORCE)
find_package(OpenSSL REQUIRED)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
set(CURL_USE_PKGCONFIG OFF CACHE BOOL "" FORCE)
set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
add_subdirectory("${CURL_SOURCE_DIRECTORY}" curl)
add_executable(native_host_http main.cpp)
target_compile_features(native_host_http PRIVATE cxx_std_17)
target_compile_definitions(native_host_http PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(native_host_http PRIVATE /utf-8 /W3 /WX)
target_link_libraries(native_host_http PRIVATE CURL::libcurl OpenSSL::SSL OpenSSL::Crypto)
